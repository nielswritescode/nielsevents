function sfValues() {
  return {
    name:      document.getElementById('sf-name').value,
    phone:     document.getElementById('sf-phone').value,
    rec:       (document.querySelector('input[name="sf-rec"]:checked') || {}).value || '',
    commit:    (document.querySelector('input[name="sf-commit"]:checked') || {}).value || '',
    sched:     document.getElementById('sf-sched').value,
    host:      document.getElementById('sf-host').value,
    questions: document.getElementById('sf-questions').value
  };
}
function sfCopy(btn) {
  function q(v) { return '"' + v.replace(/"/g, '""') + '"'; }
  var v = sfValues();
  var csv = [q(v.name), q(v.phone), q(v.rec), q(v.commit), q(v.sched), q(v.host), q(v.questions)].join(',');
  navigator.clipboard.writeText(csv).then(function() {
    btn.textContent = 'Copied!';
    setTimeout(function() { btn.textContent = 'Copy'; }, 2000);
  });
}
function sfMailto() {
  var v = sfValues();
  var body =
    'Name: ' + v.name + '\n' +
    'Phone: ' + v.phone + '\n' +
    'OK with recording: ' + v.rec + '\n' +
    'Can commit to 3/4 sessions: ' + v.commit + '\n' +
    'Scheduling needs: ' + v.sched + '\n' +
    'Hosting: ' + v.host + '\n' +
    'Questions: ' + v.questions;
  window.location.href = 'mailto:nielsevents@proton.me'
    + '?subject=' + encodeURIComponent('Signup - ' + v.name)
    + '&body=' + encodeURIComponent(body);
}
